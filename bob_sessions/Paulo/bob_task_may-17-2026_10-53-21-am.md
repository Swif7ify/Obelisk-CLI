package ui

import (

type ValidationResult struct {
    Valid   bool
    Message string
}

func ValidateAPIKey(key string) ValidationResult {
    if len(key) == 0 {
        return ValidationResult{false, "API key cannot be empty"}
    }
    if len(key) < 20 {
        return ValidationResult{false, "API key seems too short"}
    }
    if !strings.HasPrefix(key, "AI") {
        return ValidationResult{false, "Gemini API keys typically start with 'AI'"}
    }
    return ValidationResult{true, "✓ Valid format"}
}

func ValidatePath(path string) ValidationResult {
    if len(path) == 0 {
        return ValidationResult{true, "Will use current directory"}
    }
    if _, err := os.Stat(path); os.IsNotExist(err) {
        return ValidationResult{false, fmt.Sprintf("Path does not exist: %s", path)}
    }
    return ValidationResult{true, "✓ Path exists"}
}

func ValidateModel(model string) ValidationResult {
    validModels := []string{"gemini-2.5-flash", "gemini-2.5-pro", "gemini-1.5-flash"}
    for _, valid := range validModels {
        if model == valid {
            return ValidationResult{true, "✓ Valid model"}
        }
    }
    return ValidationResult{
        false, 
        fmt.Sprintf("Unknown model. Valid options: %s", strings.Join(validModels, ", ")),
    }
}

### Example 4: Progress Indicator
```go
// ui/progress.go
package ui

import (
    "fmt"
    "strings"
    "time"
)

type ProgressBar struct {
    Total   int
    Current int
    Width   int
    Start   time.Time
}

func NewProgressBar(total, width int) *ProgressBar {
    return &ProgressBar{
        Total: total,
        Width: width,
        Start: time.Now(),
    }
}

func (p *ProgressBar) Increment() {
    p.Current++
}

func (p *ProgressBar) Render() string {
    percent := float64(p.Current) / float64(p.Total)
    filled := int(percent * float64(p.Width))
    
    bar := strings.Repeat("█", filled) + strings.Repeat("░", p.Width-filled)
    
    elapsed := time.Since(p.Start)
    eta := time.Duration(0)
    if p.Current > 0 {
        eta = time.Duration(float64(elapsed) / float64(p.Current) * float64(p.Total-p.Current))
    }
    
    return fmt.Sprintf("[%s] %d/%d (%.0f%%) ETA: %s",
        bar, p.Current, p.Total, percent*100, eta.Round(time.Second))
}
```

package errors

import (
	"fmt"
	"strings"
)

// UserError represents a user-friendly error with helpful context.
type UserError struct {
	Title       string
	Description string
	Suggestion  string
	DocsLink    string
}

// Error implements the error interface.
func (e *UserError) Error() string {
	var sb strings.Builder
	
	// Title with error icon
	sb.WriteString(fmt.Sprintf("❌ %s\n\n", e.Title))
	
	// Description
	if e.Description != "" {
		sb.WriteString(fmt.Sprintf("%s\n", e.Description))
	}
	
	// Suggestion
	if e.Suggestion != "" {
		sb.WriteString(fmt.Sprintf("\n💡 Suggestion: %s\n", e.Suggestion))
	}
	
	// Documentation link
	if e.DocsLink != "" {
		sb.WriteString(fmt.Sprintf("📖 Learn more: %s\n", e.DocsLink))
	}
	
	return sb.String()
}

// NewUnsupportedProjectError creates an error for unsupported project types.
